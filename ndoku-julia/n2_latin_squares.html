function is_latin_square(adj_mtx, subset)
    check_vect = map(comb -> adj_mtx[comb...], combinations(subset, 2))
    check_vect == ones(Bool, binomial(N, 2))
end;

function find_latin_squares(adj_mtx)
    subsets = combinations(1:first(size(adj_mtx)), N)
    filter(x -> is_latin_square(adj_mtx, x), collect(subsets))
end;

@time latin_square_ids = find_latin_squares(n2_adj_mtx)

function subset_distance(dist_mtx, subset)
    sum(map(comb -> dist_mtx[comb...], combinations(subset, 2)))
end;

function subsets_distance(dist_mtx)
    subsets = combinations(1:first(size(dist_mtx)), N)
    map(x -> subset_distance(dist_mtx, x), collect(subsets))
end;

function invalid_subset_count(invalid_pair_position_mtx)
    total = 0

    for l in 1:6
        l_count = 0
        combs = collect(combinations(1:4, 2))

        for comb in combinations(1:6, l)
            res = ones(Bool, first(size(invalid_pair_position_mtx)))
            
            for col in eachcol(invalid_pair_position_mtx[:, comb])
                res = res .& col
            end

            subtotal = sum(res)
            l_count += subtotal

            # The distribution of all that has to do with the multiset 
            # permutations for the selection of pairs of positions
            # The counts seem to fall pretty well into the categories of them
            # with some counts being the same for different categories

            # See the counts by uncommenting below
            # @show (counter(vcat(combs[comb]...)), subtotal)        
        end

        total += l_count * (-1) ^ (l-1)
    end

    total
end;
